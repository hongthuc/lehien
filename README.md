      resetAutoSlide();
    }

    function goToSlide(index) {
      currentSlide = index;
      renderSlides(currentSlide);
      resetAutoSlide();
    }

    function resetAutoSlide() {
      if (slideInterval) clearInterval(slideInterval);
      slideInterval = setInterval(() => {
        currentSlide = (currentSlide + 1) % totalSlides;
        renderSlides(currentSlide);
      }, 4500);
    }

    // Bật/tắt Menu trên điện thoại
    function toggleMobileMenu() {
      const drawer = document.getElementById('mobileMenuDrawer');
      if (drawer) {
        drawer.classList.toggle('hidden');
      }
    }

    // Chọn nhanh sản phẩm đưa xuống Form Báo Giá
    function selectProductForQuote(productName) {
      const noteEl = document.getElementById('qNote');
      if (noteEl) {
        noteEl.value = 'Yêu cầu báo giá sản phẩm: ' + productName;
      }
    }

    // Xử lý tìm kiếm nhanh
    function handleSearch(event) {
      event.preventDefault();
      const dInput = document.getElementById('desktopSearchInput');
      const mInput = document.getElementById('mobileSearchInput');
      const keyword = (dInput && dInput.value.trim()) || (mInput && mInput.value.trim()) || '';
      const target = document.getElementById('danh-muc-san-pham');
      if (target) {
        target.scrollIntoView({ behavior: 'smooth' });
      }
      if (keyword) {
        selectProductForQuote(keyword);
      }
    }

    // Xử lý gửi Form Báo Giá (Không dùng alert)
    function submitQuoteForm(event) {
      event.preventDefault();
      const successBox = document.getElementById('quoteSuccessBox');
      if (successBox) {
        successBox.classList.remove('hidden');
      }
      event.target.reset();
    }

    window.onload = function () {
      resetAutoSlide();
    };
  </script>
